    SELECT BookingID, TotalAmount
    FROM BOOKING;

  v_id BOOKING.BookingID%TYPE;
  v_amount BOOKING.TotalAmount%TYPE;

  OPEN c_booking;

      DBMS_OUTPUT.PUT_LINE('Booking ' || v_id || ': High value');
    ELSE
      DBMS_OUTPUT.PUT_LINE('Booking ' || v_id || ': Normal value');
    END IF;

  END LOOP;

  CLOSE c_booking;
END;

CREATE OR REPLACE PROCEDURE proc_create_booking
(
  p_booking_id      IN NUMBER,
  p_customer_id     IN NUMBER,
  p_show_id         IN NUMBER,
  p_total_amount    IN NUMBER,
  p_add_points      IN NUMBER DEFAULT 0
)
IS
  v_status VARCHAR2(15);
BEGIN
  v_status := CASE WHEN p_total_amount > 0 THEN 'Confirmed' ELSE 'Pending' END;

  INSERT INTO BOOKING
  SELECT booking_type(p_booking_id, SYSDATE, p_total_amount, v_status, REF(c), REF(s))
  FROM CUSTOMER c, SHOW s
  WHERE c.PersonID = p_customer_id
    AND s.ShowID    = p_show_id;

  IF SQL%ROWCOUNT = 0 THEN
    DBMS_OUTPUT.PUT_LINE('X  Booking not created (customer/show not found).');
    RETURN;
  END IF;

  IF p_add_points > 0 THEN
    UPDATE CUSTOMER
    SET LoyaltyPoints = NVL(LoyaltyPoints,0) + p_add_points
    WHERE PersonID = p_customer_id;
  END IF;

  DBMS_OUTPUT.PUT_LINE(' Booking ' || p_booking_id || ' created as ' || v_status);
END;








CREATE OR REPLACE PROCEDURE proc_add_seat_to_booking
(
  p_booking_id  IN NUMBER,
  p_seat_id     IN NUMBER,
  p_price       IN NUMBER,
  p_discount    IN NUMBER DEFAULT 0
)
IS
  v_exists NUMBER;
BEGIN
  SELECT COUNT(*)
  INTO v_exists
  FROM BOOKING_SEAT
  WHERE BookingID = p_booking_id
    AND SeatID    = p_seat_id;

  IF v_exists > 0 THEN
    DBMS_OUTPUT.PUT_LINE(' Seat already added to this booking.');
    RETURN;
  END IF;

  INSERT INTO BOOKING_SEAT(BookingID, SeatID, Price, Discount_Applied)
  VALUES(p_booking_id, p_seat_id, p_price, p_discount);

  DBMS_OUTPUT.PUT_LINE(' Seat ' || p_seat_id || ' added to booking ' || p_booking_id);
END;







CREATE OR REPLACE PROCEDURE proc_recalc_booking_total
(
  p_booking_id IN NUMBER
)
IS
  v_total NUMBER := 0;
BEGIN
  SELECT NVL(SUM(NVL(Price,0) - NVL(Discount_Applied,0)),0)
  INTO v_total
  FROM BOOKING_SEAT
  WHERE BookingID = p_booking_id;

  UPDATE BOOKING
  SET TotalAmount = v_total
  WHERE BookingID = p_booking_id;

  IF SQL%ROWCOUNT = 0 THEN
    DBMS_OUTPUT.PUT_LINE(' Booking not found: ' || p_booking_id);
  ELSE
    DBMS_OUTPUT.PUT_LINE(' Booking ' || p_booking_id || ' total updated to ' || v_total);
  END IF;
END;







CREATE OR REPLACE PROCEDURE proc_record_payment
(
  p_payment_id    IN NUMBER,
  p_booking_id    IN NUMBER,
  p_method        IN VARCHAR2,
  p_discount      IN NUMBER DEFAULT 0
)
IS
  v_amount   NUMBER;
  v_status   VARCHAR2(15);
BEGIN
  SELECT TotalAmount, BookingStatus
  INTO v_amount, v_status
  FROM BOOKING
  WHERE BookingID = p_booking_id;

  IF v_status = 'Cancelled' THEN
    DBMS_OUTPUT.PUT_LINE(' Cannot pay for a Cancelled booking.');
    RETURN;
  END IF;

  INSERT INTO PAYMENT
  SELECT payment_type(p_payment_id, v_amount, p_method, SYSDATE, p_discount, REF(b))
  FROM BOOKING b
  WHERE b.BookingID = p_booking_id;

  UPDATE BOOKING
  SET BookingStatus = 'Confirmed'
  WHERE BookingID = p_booking_id;

  DBMS_OUTPUT.PUT_LINE(' Payment ' || p_payment_id || ' recorded for booking ' || p_booking_id);
END;

CREATE OR REPLACE FUNCTION fn_is_vip_show
(
  p_show_id IN NUMBER
)
RETURN VARCHAR2
IS
  v_count NUMBER;
BEGIN
  SELECT COUNT(*)
  INTO v_count
  FROM SEAT s
  JOIN SHOW sh
    ON DEREF(s.Hall).HallID = DEREF(sh.Hall).HallID
  WHERE sh.ShowID = p_show_id
    AND s.SeatType = 'VIP';

  IF v_count > 0 THEN
    RETURN 'YES';
  ELSE
    RETURN 'NO';
  END IF;
END;

  );
END;

CREATE OR REPLACE PACKAGE BODY booking_pkg AS

  PROCEDURE create_booking(
    p_booking_id  IN NUMBER,
    p_customer_id IN NUMBER,
    p_show_id     IN NUMBER
  )
  IS
    v_cust REF customer_type;
    v_show REF show_type;
  BEGIN
    -- Get customer REF (predefined exception: NO_DATA_FOUND if missing)
    SELECT REF(c)
    INTO v_cust
    FROM CUSTOMER c
    WHERE c.PersonID = p_customer_id;

    -- Get show REF via view (avoid SHOW keyword issues)
    SELECT REF(s)
    INTO v_show
    FROM SHOW_T s
    WHERE s.ShowID = p_show_id;

    -- Insert into object table safely
    INSERT INTO BOOKING
    SELECT booking_type(
      p_booking_id,
      SYSDATE,
      0,
      'Pending',
      v_cust,
      v_show
    )
    FROM dual;

    DBMS_OUTPUT.PUT_LINE('Booking created: ' || p_booking_id);

  EXCEPTION
    WHEN NO_DATA_FOUND THEN
      RAISE_APPLICATION_ERROR(-20001, 'Customer or Show not found');
    WHEN DUP_VAL_ON_INDEX THEN
      RAISE_APPLICATION_ERROR(-20002, 'Booking ID already exists');
  END create_booking;


  PROCEDURE add_seat(
    p_booking_id IN NUMBER,
    p_seat_id    IN NUMBER,
    p_price      IN NUMBER,
    p_discount   IN NUMBER DEFAULT 0
  )
  IS
    v_exists NUMBER;
    v_dummy  NUMBER;
  BEGIN
    -- Ensure booking exists (NO_DATA_FOUND if missing)
    SELECT 1 INTO v_dummy
    FROM BOOKING b
    WHERE b.BookingID = p_booking_id;

    -- Ensure seat exists (NO_DATA_FOUND if missing)
    SELECT 1 INTO v_dummy
    FROM SEAT s
    WHERE s.SeatID = p_seat_id;

    -- Prevent duplicates (user-defined exception)
    SELECT COUNT(*)
    INTO v_exists
    FROM BOOKING_SEAT
    WHERE BookingID = p_booking_id
      AND SeatID    = p_seat_id;

    IF v_exists > 0 THEN
      RAISE_APPLICATION_ERROR(-20003, 'Seat already added to this booking');
    END IF;

    INSERT INTO BOOKING_SEAT(BookingID, SeatID, Price, Discount_Applied)
    VALUES(p_booking_id, p_seat_id, p_price, p_discount);

    DBMS_OUTPUT.PUT_LINE('Seat added: ' || p_seat_id || ' to booking ' || p_booking_id);

  EXCEPTION
    WHEN NO_DATA_FOUND THEN
      RAISE_APPLICATION_ERROR(-20004, 'Booking or Seat not found');
  END add_seat;


  PROCEDURE recalc_total(
    p_booking_id IN NUMBER
  )
  IS
    v_total NUMBER;
  BEGIN
    SELECT NVL(SUM(NVL(Price,0) - NVL(Discount_Applied,0)), 0)
    INTO v_total
    FROM BOOKING_SEAT
    WHERE BookingID = p_booking_id;

    UPDATE BOOKING
    SET TotalAmount = v_total
    WHERE BookingID = p_booking_id;

    IF SQL%ROWCOUNT = 0 THEN
      RAISE_APPLICATION_ERROR(-20005, 'Booking not found');
    END IF;

    DBMS_OUTPUT.PUT_LINE('Total updated for booking ' || p_booking_id || ' = ' || v_total);
  END recalc_total;


  FUNCTION booking_total(
    p_booking_id IN NUMBER
  )
  RETURN NUMBER
  IS
    v_total NUMBER;
  BEGIN
    SELECT NVL(SUM(NVL(Price,0) - NVL(Discount_Applied,0)), 0)
    INTO v_total
    FROM BOOKING_SEAT
    WHERE BookingID = p_booking_id;

    RETURN v_total;
  END booking_total;

END booking_pkg;
